#面试题 

### 代码演示
- 支持`Set`、`Map`
- 能处理循环引用的数据

```js
// 复杂版本, 使用weakMap,弱引入,不影响垃圾回收
function deepClone(target, map = new WeakMap()){
  if(typeof target !== 'object' && target !== null) return target;

  // 处理循环引用的数据
  const repeatData = map.get(target);
  if(repeatData) return repeatData;

  // 初始化
  let results = {};
  // 把target记录在map当中
  map.set(target, results);

  // Set
  if(target instanceof Set){
    results = new Set();
    target.forEach(v=>{
      // set中的复杂数据也需要深度克隆
      const value = deepClone(v, map);
      results.add(value)
    })
  }

  // Map
  if(target instanceof Map){
    results = new Map();
    target.forEach((v1, v2)=>{
      // Map 中的复杂数据也需要深度克隆
      const key = deepClone(v2, map);
      const value = deepClone(v1, map);

      results.set(key, value);
    })
  }


  // 数组
  if(target instanceof Array){
    return results = target.map(v => deepClone(v, map))
  }

  // 对象
  for (const key in target) {
    if(Object.hasOwnProperty.call(target, key)){
      results[key] = deepClone(target[key], map);
    }
  }

  return results;
}

const temp = {
  a: 1
};

const temp2 = {
  b: 2,
  a: temp
}

console.log(deepClone({
  foo: temp,
  bar: [1, 2, 3, 4],
  baz: new Set([1,2,3,4]),
  buz: new Map().set('id', 123),
  doc: {
    temp2,
  }
})); // { foo: { a: 1 }, bar: [ 1, 2, 3, 4 ], baz: Set(4) { 1, 2, 3, 4 }, buz: Map(1) { 'id' => 123 }, doc: { temp2: { b: 2, a: [Object] } }
}
```
